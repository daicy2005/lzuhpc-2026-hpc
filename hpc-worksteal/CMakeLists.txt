cmake_minimum_required(VERSION 3.16)
project(mini_worksteal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json 3.11 REQUIRED)
option(WS_ASAN "Address and undefined behavior sanitizers" OFF)
option(WS_TSAN "Thread sanitizer" OFF)
option(WS_ENABLE_IPO "Enable checked interprocedural optimization" OFF)
option(WS_STEAL_SCRATCH "Reuse per-worker steal storage and avoid single-task allocation" OFF)
set(WS_COMPLETION_BATCH 1 CACHE STRING "Task completions published per atomic update")
set(WS_COMPUTE_LANES 1 CACHE STRING "Independent compute elements interleaved per loop (1,4,8)")
if(NOT WS_COMPUTE_LANES MATCHES "^(1|4|8)$")
  message(FATAL_ERROR "WS_COMPUTE_LANES must be 1,4 or 8")
endif()
if(NOT WS_COMPLETION_BATCH MATCHES "^[1-9][0-9]*$" OR WS_COMPLETION_BATCH GREATER 1024)
  message(FATAL_ERROR "WS_COMPLETION_BATCH must be an integer in [1,1024]")
endif()
if(WS_ENABLE_IPO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT WS_IPO_SUPPORTED OUTPUT WS_IPO_ERROR)
  if(NOT WS_IPO_SUPPORTED)
    message(FATAL_ERROR "IPO unavailable: ${WS_IPO_ERROR}")
  endif()
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
if(WS_ASAN AND WS_TSAN)
  message(FATAL_ERROR "Sanitizers must use separate builds")
endif()
add_library(ws_core src/runtime/worker_pool.cpp src/workloads/workload.cpp)
target_compile_definitions(ws_core PRIVATE WS_COMPLETION_BATCH=${WS_COMPLETION_BATCH} WS_STEAL_SCRATCH=$<BOOL:${WS_STEAL_SCRATCH}> WS_COMPUTE_LANES=${WS_COMPUTE_LANES})
target_include_directories(ws_core PUBLIC include)
target_link_libraries(ws_core PUBLIC pthread nlohmann_json::nlohmann_json)
target_compile_options(ws_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion)
if(WS_ASAN)
  target_compile_options(ws_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ws_core PUBLIC -fsanitize=address,undefined)
elseif(WS_TSAN)
  target_compile_options(ws_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(ws_core PUBLIC -fsanitize=thread)
endif()
set(WS_REGISTRATIONS src/policies/register_defaults.cpp)
add_executable(worksteal src/app/main.cpp ${WS_REGISTRATIONS})
target_link_libraries(worksteal PRIVATE ws_core)
enable_testing()
add_executable(runtime_checks tests/runtime_checks.cpp ${WS_REGISTRATIONS})
target_link_libraries(runtime_checks PRIVATE ws_core)
add_executable(worksteal_extension src/app/main.cpp ${WS_REGISTRATIONS} tests/register_reverse.cpp)
target_link_libraries(worksteal_extension PRIVATE ws_core)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME runtime_checks COMMAND runtime_checks)
set_tests_properties(runtime_checks PROPERTIES TIMEOUT 60)
add_test(NAME cli_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_checks.py $<TARGET_FILE:worksteal> $<TARGET_FILE:worksteal_extension>)
set_tests_properties(cli_checks PROPERTIES TIMEOUT 120)
add_executable(host_probe src/platform/probe.cpp ${WS_REGISTRATIONS})
target_link_libraries(host_probe PRIVATE ws_core)
